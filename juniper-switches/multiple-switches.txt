---
- name: Propagate VLAN to all existing trunk ports on selected Junos switches
  hosts: junos_switches
  gather_facts: false

  vars:
    vlan_name: "vlan{{ vlan_id }}"
    vlan_description: "Propagated by Ansible"

  pre_tasks:
    - name: Validate required variables
      ansible.builtin.assert:
        that:
          - vlan_id is defined
          - vlan_id | int >= 1
          - vlan_id | int <= 4094
          - vlan_name is defined
        fail_msg: >
          You must provide vlan_id and vlan_name.
          Example: -e "vlan_id=126 vlan_name=vlan126"

  tasks:
    - name: Backup current Junos configuration
      junipernetworks.junos.junos_config:
        backup: true
      register: backup_result

    - name: Ensure VLAN exists on switch
      junipernetworks.junos.junos_vlans:
        config:
          - name: "{{ vlan_name }}"
            vlan_id: "{{ vlan_id | int }}"
            description: "{{ vlan_description }}"
        state: merged

    - name: Get interface configuration in set format
      junipernetworks.junos.junos_command:
        commands:
          - show configuration interfaces
        display: set
      register: interfaces_config

    - name: Convert interface config output to text
      ansible.builtin.set_fact:
        interfaces_config_text: "{{ interfaces_config.stdout_lines[0] | default([]) | join('\n') }}"

    - name: Detect trunk ports
      ansible.builtin.set_fact:
        trunk_ports: >-
          {{
            interfaces_config_text
            | regex_findall('set interfaces\\s+(\\S+)\\s+unit\\s+(\\d+)\\s+family\\s+ethernet-switching\\s+(?:interface-mode|port-mode)\\s+trunk')
            | unique
            | list
          }}

    - name: Detect trunk ports that already allow all VLANs
      ansible.builtin.set_fact:
        trunk_ports_with_all_vlans: >-
          {{
            interfaces_config_text
            | regex_findall('set interfaces\\s+(\\S+)\\s+unit\\s+(\\d+)\\s+family\\s+ethernet-switching\\s+vlan\\s+members\\s+all')
            | unique
            | list
          }}

    - name: Build final trunk port list to update
      ansible.builtin.set_fact:
        trunk_ports_to_update: "{{ trunk_ports | difference(trunk_ports_with_all_vlans) }}"

    - name: Show detected trunk ports
      ansible.builtin.debug:
        msg:
          - "Switch: {{ inventory_hostname }}"
          - "Detected trunk ports: {{ trunk_ports }}"
          - "Trunks with vlan members all, skipped: {{ trunk_ports_with_all_vlans }}"
          - "Trunks to update: {{ trunk_ports_to_update }}"

    - name: Build configuration lines for trunk ports
      ansible.builtin.set_fact:
        trunk_config_lines: "{{ trunk_config_lines | default([]) + [ 'set interfaces ' ~ item.0 ~ ' unit ' ~ item.1 ~ ' family ethernet-switching vlan members ' ~ vlan_name ] }}"
      loop: "{{ trunk_ports_to_update }}"
      when: trunk_ports_to_update | length > 0

    - name: Add VLAN to all detected trunk ports
      junipernetworks.junos.junos_config:
        lines: "{{ trunk_config_lines }}"
        comment: "Propagated {{ vlan_name }} to trunk ports using Ansible"
      when:
        - trunk_ports_to_update | length > 0
        - trunk_config_lines is defined

    - name: No trunk ports found or no update needed
      ansible.builtin.debug:
        msg: "No trunk ports required update on {{ inventory_hostname }}."
      when: trunk_ports_to_update | length == 0

    - name: Show backup path
      ansible.builtin.debug:
        msg: "Backup for {{ inventory_hostname }} saved at {{ backup_result.backup_path | default('N/A') }}"
